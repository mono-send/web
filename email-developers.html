ci/send-welcome.ts
// ship via CI with typed SDK
import { Monosend } from 'monosend';
const client = new Monosend(process.env.MONO_API_KEY);

await client.emails.send({
  from: 'dev@product.io',
  to: 'user@inbox.dev',
  subject: 'Welcome!',
  html: `<EmailPreview />`,
});

// assertions inside tests
await client.emails.assertDelivered('user@inbox.dev');

name: deploy-emails
on: [push]

  deliverability:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - run: monosend preview --ci
      - run: monosend test --assert-delivery
      - run: monosend deploy --env=prod
